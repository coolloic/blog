{{- /* The articles of a topic, in the order given by its `articles` front matter. Unknown slugs fail the build. */ -}} {{- $out := slice -}} {{- range .Params.articles }} {{- with site.GetPage (printf "/articles/%s" .) }}{{ $out = $out | append . }} {{- else }}{{ errorf "%s: unknown article %q in articles" $.File.Path . }}{{ end }} {{- end -}} {{- if lt (len $out) 2 }}{{ errorf "%s: a topic needs at least 2 articles" .File.Path }}{{ end -}} {{- return $out -}}
