{{- /* Resolves relative links to .md files (e.g. [MCP](mcp.md), [x](../sources/y.md#part)) to the target page's URL, so the same Markdown works in Obsidian and on the site. An unresolvable .md link emits a warning; build with --panicOnWarning to fail on it. */ -}} {{- $u := urls.Parse .Destination -}} {{- $href := .Destination -}}{{- $resolved := false -}} {{- if and (not $u.IsAbs) (strings.HasSuffix $u.Path ".md") -}} {{- with .PageInner.GetPage $u.Path -}} {{- $href = .RelPermalink -}}{{- $resolved = true -}} {{- with $u.Fragment }}{{ $href = printf "%s#%s" $href . }}{{ end -}} {{- else -}} {{- warnf "%s: broken wiki link %q" $.PageInner.File.Path $.Destination -}} {{- end -}} {{- end -}} {{- /* only trust URLs we built; others go through html/template sanitising (blocks javascript: etc.) */ -}} {{ .Text }} {{- /* chomp trailing newline */ -}}
